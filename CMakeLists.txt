cmake_minimum_required (VERSION 3.8)
project(project_PyroProgression)
option(DISCOVERY_LOCK_DEFAULT "Lock region baselines on first entry" ON)
add_subdirectory(CWSDK)
add_library (PyroProgression SHARED "main.cpp" "src/XPOverwrite.h"  "src/LevelDisplayOverwrite.h" "src/utility.h" "src/utility.cpp" "src/GearScalingOverWrite.h" "src/GoldDropOverWrite.h")
target_link_libraries (PyroProgression LINK_PUBLIC CWSDK)
target_sources(PyroProgression PRIVATE src/RelativeScaling.cpp)
target_compile_features(PyroProgression PUBLIC cxx_std_17)
target_compile_definitions(PyroProgression PRIVATE NOMINMAX DISCOVERY_LOCK_DEFAULT=$<BOOL:${DISCOVERY_LOCK_DEFAULT}>)
target_compile_options(PyroProgression PRIVATE /clang:-fasm-blocks)

include(CTest)
if(BUILD_TESTING)
  add_executable(RelativeRulesTest RulesTest.cpp)
  target_compile_features(RelativeRulesTest PRIVATE cxx_std_17)
  target_compile_options(RelativeRulesTest PRIVATE /UNDEBUG)
  add_test(NAME RelativeRules COMMAND RelativeRulesTest)
endif()
